/**
 * LEVEL 1 — Standard EPA SWMM5 Binary .out File Parser
 * Reads the 34 standard variables from the binary output format.
 *
 * Binary layout:
 *   HEADER: magic(4), version(4), flowUnits(4), nSubcatch(4), nNodes(4), nLinks(4), nPolluts(4)
 *   ID RECORDS: subcatch names, node names, link names, pollutant names
 *   PROPERTY RECORDS: subcatch props, node props (type, invert, maxDepth), link props (type, offsets, etc.)
 *   REPORTING VARIABLES: subcatch var codes, node var codes, link var codes, system var codes
 *   START DATE (double), REPORT STEP (int)
 *   RESULTS: [date(double) + subcatch(float[]) + node(float[]) + link(float[]) + system(float[])] × nPeriods
 *   FOOTER: IDpos(4), ObjPropPos(4), ResultsPos(4), nPeriods(4), errCode(4), magic(4)
 */

// Standard EPA result variable names
export const NODE_VARS = ['depth', 'head', 'volume', 'latInflow', 'totalInflow', 'flooding'];
export const LINK_VARS = ['flow', 'depth', 'velocity', 'volume', 'capacity'];
export const SUBCATCH_VARS = ['rainfall', 'snowDepth', 'evap', 'infil', 'runoff', 'gwFlow', 'gwElev', 'soilMoisture'];
export const SYSTEM_VARS = [
  'airTemp', 'rainfall', 'snowDepth', 'evapLoss', 'runoff',
  'dwInflow', 'gwInflow', 'rdiiInflow', 'directInflow',
  'totalLatInflow', 'flooding', 'outflow', 'storageVol',
  'evaporation', 'seepageLoss'
];

// Node and link property indices in the .out file
export const NODE_TYPES = { 0: 'junction', 1: 'outfall', 2: 'storage', 3: 'divider' };
export const LINK_TYPES = { 0: 'conduit', 1: 'pump', 2: 'orifice', 3: 'weir', 4: 'outlet' };

export class SWMMOutputLevel1 {
  constructor(arrayBuffer) {
    this.buffer = arrayBuffer;
    this.view = new DataView(arrayBuffer);
    this.offset = 0;

    // Metadata
    this.version = 0;
    this.flowUnits = 0;
    this.nSubcatch = 0;
    this.nNodes = 0;
    this.nLinks = 0;
    this.nPolluts = 0;
    this.nPeriods = 0;
    this.reportStep = 300;
    this.startDate = 0;

    // Object names
    this.subcatchNames = [];
    this.nodeNames = [];
    this.linkNames = [];
    this.pollutNames = [];

    // Object properties from header
    this.nodeTypes = {};
    this.nodeInverts = {};
    this.nodeMaxDepths = {};
    this.linkTypes = {};
    this.linkOffsets1 = {};
    this.linkOffsets2 = {};
    this.linkMaxDepths = {};

    // Reporting variable counts
    this.nSubcatchVars = 0;
    this.nNodeVars = 0;
    this.nLinkVars = 0;
    this.nSystemVars = 0;

    // Results data pointer
    this.resultOffset = 0;
    this.bytesPerPeriod = 0;

    // Cache for performance
    this._cache = new Map();
    this._maxCacheSize = 50000;

    this._parse();
  }

  // ─── BINARY READERS ──────────────────────────────

  _readInt(offset) {
    if (offset !== undefined) return this.view.getInt32(offset, true);
    const val = this.view.getInt32(this.offset, true);
    this.offset += 4;
    return val;
  }

  _readFloat(offset) {
    if (offset !== undefined) return this.view.getFloat32(offset, true);
    const val = this.view.getFloat32(this.offset, true);
    this.offset += 4;
    return val;
  }

  _readDouble(offset) {
    if (offset !== undefined) return this.view.getFloat64(offset, true);
    const val = this.view.getFloat64(this.offset, true);
    this.offset += 8;
    return val;
  }

  _readString() {
    const len = this._readInt();
    if (len <= 0 || len > 256) return '';
    const bytes = new Uint8Array(this.buffer, this.offset, len);
    this.offset += len;
    let str = '';
    for (let i = 0; i < len; i++) {
      if (bytes[i] === 0) break;
      str += String.fromCharCode(bytes[i]);
    }
    return str;
  }

  // ─── PARSER ──────────────────────────────────────

  _parse() {
    try {
      this.offset = 0;

      // ── HEADER ──
      const magic1 = this._readInt();
      this.version = this._readInt();
      this.flowUnits = this._readInt();
      this.nSubcatch = this._readInt();
      this.nNodes = this._readInt();
      this.nLinks = this._readInt();
      this.nPolluts = this._readInt();

      // ── ID STRINGS ──
      for (let i = 0; i < this.nSubcatch; i++) this.subcatchNames.push(this._readString());
      for (let i = 0; i < this.nNodes; i++) this.nodeNames.push(this._readString());
      for (let i = 0; i < this.nLinks; i++) this.linkNames.push(this._readString());
      for (let i = 0; i < this.nPolluts; i++) this.pollutNames.push(this._readString());

      // ── OBJECT PROPERTIES ──
      // Subcatchment properties
      const nSubcatchProps = this._readInt();
      for (let i = 0; i < nSubcatchProps; i++) this._readInt(); // prop codes
      for (let i = 0; i < this.nSubcatch; i++) {
        for (let j = 0; j < nSubcatchProps; j++) this._readFloat();
      }

      // Node properties
      const nNodeProps = this._readInt();
      const nodePropCodes = [];
      for (let i = 0; i < nNodeProps; i++) nodePropCodes.push(this._readInt());
      for (let i = 0; i < this.nNodes; i++) {
        const props = [];
        for (let j = 0; j < nNodeProps; j++) props.push(this._readFloat());
        const name = this.nodeNames[i];
        if (nNodeProps >= 1) this.nodeTypes[name] = Math.round(props[0]);
        if (nNodeProps >= 2) this.nodeInverts[name] = props[1];
        if (nNodeProps >= 3) this.nodeMaxDepths[name] = props[2];
      }

      // Link properties
      const nLinkProps = this._readInt();
      const linkPropCodes = [];
      for (let i = 0; i < nLinkProps; i++) linkPropCodes.push(this._readInt());
      for (let i = 0; i < this.nLinks; i++) {
        const props = [];
        for (let j = 0; j < nLinkProps; j++) props.push(this._readFloat());
        const name = this.linkNames[i];
        if (nLinkProps >= 1) this.linkTypes[name] = Math.round(props[0]);
        if (nLinkProps >= 2) this.linkOffsets1[name] = props[1];
        if (nLinkProps >= 3) this.linkOffsets2[name] = props[2];
        if (nLinkProps >= 5) this.linkMaxDepths[name] = props[4];
      }

      // ── REPORTING VARIABLES ──
      this.nSubcatchVars = this._readInt();
      for (let i = 0; i < this.nSubcatchVars; i++) this._readInt();
      this.nNodeVars = this._readInt();
      for (let i = 0; i < this.nNodeVars; i++) this._readInt();
      this.nLinkVars = this._readInt();
      for (let i = 0; i < this.nLinkVars; i++) this._readInt();
      this.nSystemVars = this._readInt();
      for (let i = 0; i < this.nSystemVars; i++) this._readInt();

      // ── START DATE & REPORT STEP ──
      this.startDate = this._readDouble();
      this.reportStep = this._readInt();

      // ── RESULTS OFFSET ──
      this.resultOffset = this.offset;

      // Calculate bytes per reporting period
      this.bytesPerPeriod = 8 + // date double
        (this.nSubcatch * this.nSubcatchVars * 4) +
        (this.nNodes * this.nNodeVars * 4) +
        (this.nLinks * this.nLinkVars * 4) +
        (this.nSystemVars * 4);

      // ── READ FOOTER ──
      const footerSize = 6 * 4; // 6 integers
      if (this.buffer.byteLength >= footerSize) {
        const footerStart = this.buffer.byteLength - footerSize;
        // Footer: IDpos, ObjPropPos, ResultsPos, nPeriods, errCode, magic
        this.nPeriods = this._readInt(footerStart + 12);
        const errCode = this._readInt(footerStart + 16);
        const magic2 = this._readInt(footerStart + 20);

        // Validate
        if (this.nPeriods <= 0 || this.nPeriods > 1000000) {
          const dataBytes = this.buffer.byteLength - this.resultOffset - footerSize;
          if (this.bytesPerPeriod > 0) {
            this.nPeriods = Math.floor(dataBytes / this.bytesPerPeriod);
          }
        }
      }

      console.log(`SWMM Output: ${this.nSubcatch} subcatch, ${this.nNodes} nodes, ${this.nLinks} links, ${this.nPeriods} periods, step=${this.reportStep}s`);
    } catch (e) {
      console.error('Error parsing SWMM .out file:', e);
    }
  }

  // ─── DATA ACCESS ─────────────────────────────────

  _getResultOffset(period, subcatchIdx, nodeIdx, linkIdx, varIdx) {
    if (period < 0 || period >= this.nPeriods) return -1;
    let offset = this.resultOffset + period * this.bytesPerPeriod;
    offset += 8; // skip date double

    if (subcatchIdx !== null && subcatchIdx !== undefined) {
      offset += (subcatchIdx * this.nSubcatchVars + varIdx) * 4;
    } else if (nodeIdx !== null && nodeIdx !== undefined) {
      offset += this.nSubcatch * this.nSubcatchVars * 4;
      offset += (nodeIdx * this.nNodeVars + varIdx) * 4;
    } else if (linkIdx !== null && linkIdx !== undefined) {
      offset += this.nSubcatch * this.nSubcatchVars * 4;
      offset += this.nNodes * this.nNodeVars * 4;
      offset += (linkIdx * this.nLinkVars + varIdx) * 4;
    } else {
      // System variable
      offset += this.nSubcatch * this.nSubcatchVars * 4;
      offset += this.nNodes * this.nNodeVars * 4;
      offset += this.nLinks * this.nLinkVars * 4;
      offset += varIdx * 4;
    }

    return offset;
  }

  _getCachedFloat(offset) {
    if (this._cache.has(offset)) return this._cache.get(offset);
    try {
      const val = this.view.getFloat32(offset, true);
      if (this._cache.size < this._maxCacheSize) {
        this._cache.set(offset, val);
      }
      return val;
    } catch {
      return 0;
    }
  }

  getNodeResult(nodeId, variable, timestep) {
    const idx = this.nodeNames.indexOf(nodeId);
    if (idx < 0) return 0;
    const varIdx = typeof variable === 'number' ? variable : NODE_VARS.indexOf(variable);
    if (varIdx < 0 || varIdx >= this.nNodeVars) return 0;
    const offset = this._getResultOffset(timestep, null, idx, null, varIdx);
    return offset >= 0 ? this._getCachedFloat(offset) : 0;
  }

  getLinkResult(linkId, variable, timestep) {
    const idx = this.linkNames.indexOf(linkId);
    if (idx < 0) return 0;
    const varIdx = typeof variable === 'number' ? variable : LINK_VARS.indexOf(variable);
    if (varIdx < 0 || varIdx >= this.nLinkVars) return 0;
    const offset = this._getResultOffset(timestep, null, null, idx, varIdx);
    return offset >= 0 ? this._getCachedFloat(offset) : 0;
  }

  getSubcatchResult(subcatchId, variable, timestep) {
    const idx = this.subcatchNames.indexOf(subcatchId);
    if (idx < 0) return 0;
    const varIdx = typeof variable === 'number' ? variable : SUBCATCH_VARS.indexOf(variable);
    if (varIdx < 0 || varIdx >= this.nSubcatchVars) return 0;
    const offset = this._getResultOffset(timestep, idx, null, null, varIdx);
    return offset >= 0 ? this._getCachedFloat(offset) : 0;
  }

  getSystemResult(variable, timestep) {
    const varIdx = typeof variable === 'number' ? variable : SYSTEM_VARS.indexOf(variable);
    if (varIdx < 0 || varIdx >= this.nSystemVars) return 0;
    if (timestep < 0 || timestep >= this.nPeriods) return 0;
    const offset = this._getResultOffset(timestep, null, null, null, varIdx);
    return offset >= 0 ? this._getCachedFloat(offset) : 0;
  }

  // ─── TIME SERIES EXTRACTION ──────────────────────

  getNodeSeries(nodeId, variable) {
    const series = [];
    for (let t = 0; t < this.nPeriods; t++) {
      series.push({
        time: t * this.reportStep,
        timeStr: this._formatTime(t),
        value: this.getNodeResult(nodeId, variable, t),
      });
    }
    return series;
  }

  getLinkSeries(linkId, variable) {
    const series = [];
    for (let t = 0; t < this.nPeriods; t++) {
      series.push({
        time: t * this.reportStep,
        timeStr: this._formatTime(t),
        value: this.getLinkResult(linkId, variable, t),
      });
    }
    return series;
  }

  getSubcatchSeries(subcatchId, variable) {
    const series = [];
    for (let t = 0; t < this.nPeriods; t++) {
      series.push({
        time: t * this.reportStep,
        timeStr: this._formatTime(t),
        value: this.getSubcatchResult(subcatchId, variable, t),
      });
    }
    return series;
  }

  getSystemSeries(variable) {
    const series = [];
    for (let t = 0; t < this.nPeriods; t++) {
      series.push({
        time: t * this.reportStep,
        timeStr: this._formatTime(t),
        value: this.getSystemResult(variable, t),
      });
    }
    return series;
  }

  // ─── STATISTICS ──────────────────────────────────

  getNodeStats(nodeId) {
    const stats = {};
    for (const varName of NODE_VARS) {
      let min = Infinity, max = -Infinity, sum = 0, maxT = 0;
      for (let t = 0; t < this.nPeriods; t++) {
        const v = this.getNodeResult(nodeId, varName, t);
        if (v < min) min = v;
        if (v > max) { max = v; maxT = t; }
        sum += v;
      }
      stats[varName] = {
        min: min === Infinity ? 0 : min,
        max: max === -Infinity ? 0 : max,
        mean: this.nPeriods > 0 ? sum / this.nPeriods : 0,
        timeOfMax: maxT,
        timeOfMaxStr: this._formatTime(maxT),
      };
    }
    return stats;
  }

  getLinkStats(linkId) {
    const stats = {};
    for (const varName of LINK_VARS) {
      let min = Infinity, max = -Infinity, sum = 0, maxT = 0;
      for (let t = 0; t < this.nPeriods; t++) {
        const v = this.getLinkResult(linkId, varName, t);
        const absV = Math.abs(v);
        if (v < min) min = v;
        if (absV > Math.abs(max)) { max = v; maxT = t; }
        sum += absV;
      }
      stats[varName] = {
        min, max,
        mean: this.nPeriods > 0 ? sum / this.nPeriods : 0,
        timeOfMax: maxT,
        timeOfMaxStr: this._formatTime(maxT),
      };
    }
    return stats;
  }

  // ─── LEGEND RANGES ───────────────────────────────

  getLegendRange(objectType, variable, timestep) {
    let min = Infinity, max = -Infinity;
    const names = objectType === 'node' ? this.nodeNames :
                  objectType === 'link' ? this.linkNames :
                  this.subcatchNames;
    const getter = objectType === 'node' ? 'getNodeResult' :
                   objectType === 'link' ? 'getLinkResult' :
                   'getSubcatchResult';

    for (const name of names) {
      const val = this[getter](name, variable, timestep);
      if (val < min) min = val;
      if (val > max) max = val;
    }

    if (min === Infinity) min = 0;
    if (max === -Infinity) max = 1;
    if (Math.abs(max - min) < 1e-10) max = min + 1;
    return { min, max };
  }

  // ─── SUMMARY ─────────────────────────────────────

  getTimesteps() {
    return Array.from({ length: this.nPeriods }, (_, i) => this._formatTime(i));
  }

  getSummary() {
    return {
      version: this.version,
      flowUnits: this.flowUnits,
      nSubcatch: this.nSubcatch,
      nNodes: this.nNodes,
      nLinks: this.nLinks,
      nPolluts: this.nPolluts,
      nPeriods: this.nPeriods,
      reportStep: this.reportStep,
      subcatchNames: this.subcatchNames,
      nodeNames: this.nodeNames,
      linkNames: this.linkNames,
      pollutNames: this.pollutNames,
      nodeTypes: this.nodeTypes,
      linkTypes: this.linkTypes,
      nSubcatchVars: this.nSubcatchVars,
      nNodeVars: this.nNodeVars,
      nLinkVars: this.nLinkVars,
      nSystemVars: this.nSystemVars,
    };
  }

  _formatTime(timestep) {
    const totalSec = timestep * this.reportStep;
    const h = Math.floor(totalSec / 3600);
    const m = Math.floor((totalSec % 3600) / 60);
    const s = totalSec % 60;
    return`${h.toString().padStart(2, '0')}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
  }
}

/**
 * LEVEL 2 — Hydraulic Diagnostics Computed from Standard Output
 *
 * Computes ~40 derived variables from the 34 standard Level 1 variables:
 *   - Froude number, flow classification, normal depth/flow
 *   - Friction slope, head losses, energy grade line
 *   - Pipe slopes, capacity ratios, self-cleansing velocity checks
 *   - Cross-section geometry (area, hydraulic radius, top width) for 10+ shapes
 *   - Bernoulli energy balance
 *   - System-wide QA metrics
 *
 * Does NOT require a modified engine — works with any EPA SWMM5 .out file.
 */

const GRAVITY = 32.174; // ft/s² (US customary)
const GRAVITY_SI = 9.81; // m/s² (SI)
const PHI_US = 1.4859; // Manning's κ = 1.49 for US units
const PHI_SI = 1.0;

export class SWMMOutputLevel2 {
  constructor(level1, model) {
    this.L1 = level1; // SWMMOutputLevel1 instance
    this.model = model; // SWMMModel instance
    this.g = GRAVITY; // default US
    this.phi = PHI_US;

    // Detect SI units
    if (model && model.options) {
      const units = (model.options.FLOW_UNITS || '').toUpperCase();
      if (['CMS', 'LPS', 'MLD'].includes(units)) {
        this.g = GRAVITY_SI;
        this.phi = PHI_SI;
      }
    }

    // Pre-compute static properties
    this._conduitData = new Map();
    this._precomputeConduitData();
  }

  _precomputeConduitData() {
    if (!this.model) return;
    this.model.conduits.forEach((c, id) => {
      const fromNode = this._getNode(c.fromNode);
      const toNode = this._getNode(c.toNode);
      const invertUp = (fromNode?.invertElev || 0) + (c.inletOffset || 0);
      const invertDn = (toNode?.invertElev || 0) + (c.outletOffset || 0);
      const length = c.length || 1;
      const slope = (invertUp - invertDn) / length;
      const n = c.roughness || 0.013;
      const shape = (c.xsecShape || 'CIRCULAR').toUpperCase();
      const geom1 = c.xsecGeom1 || 1;
      const geom2 = c.xsecGeom2 || 0;
      const geom3 = c.xsecGeom3 || 0;
      const geom4 = c.xsecGeom4 || 0;
      const barrels = c.xsecBarrels || 1;
      const entryLoss = c.entryLoss || 0;
      const exitLoss = c.exitLoss || 0;
      const avgLoss = c.avgLoss || 0;

      const aFull = this._getAreaFull(shape, geom1, geom2);
      const rFull = this._getHydRadiusFull(shape, geom1, geom2);
      const qFull = (this.phi / n) * aFull * Math.pow(rFull, 2 / 3) * Math.sqrt(Math.abs(slope));

      this._conduitData.set(id, {
        fromNode: c.fromNode, toNode: c.toNode,
        invertUp, invertDn, length, slope, n,
        shape, geom1, geom2, geom3, geom4, barrels,
        entryLoss, exitLoss, avgLoss,
        aFull, rFull, qFull,
      });
    });
  }

  _getNode(id) {
    if (!this.model) return null;
    return this.model.junctions.get(id) ||
           this.model.outfalls.get(id) ||
           this.model.storage.get(id) ||
           this.model.dividers.get(id);
  }

  // ═══════════════════════════════════════════════════
  // CROSS-SECTION GEOMETRY — for any depth and shape
  // ═══════════════════════════════════════════════════

  getArea(shape, geom1, geom2, depth) {
    const d = Math.max(0, Math.min(depth, geom1 || 1));
    const D = geom1 || 1;

    switch (shape) {
      case 'CIRCULAR': {
        if (d <= 0) return 0;
        if (d >= D) return Math.PI * D * D / 4;
        const r = D / 2;
        const ratio = Math.max(-1, Math.min(1, 1 - d / r));
        const theta = 2 * Math.acos(ratio);
        return r * r / 2 * (theta - Math.sin(theta));
      }
      case 'RECT_CLOSED':
      case 'RECT_OPEN': {
        const w = geom2 || D;
        return d * w;
      }
      case 'TRAPEZOIDAL': {
        const botW = geom2 || D * 0.5;
        const sideSlope = geom2 || 1;
        return d * (botW + sideSlope * d);
      }
      case 'TRIANGULAR': {
        const sideSlope = geom2 || 1;
        return sideSlope * d * d;
      }
      case 'HORIZ_ELLIPSE': {
        const a = (geom2 || D * 1.5) / 2; // semi-major
        const b = D / 2; // semi-minor
        if (d >= D) return Math.PI * a * b;
        const ratio = (d - b) / b;
        const t = Math.asin(Math.max(-1, Math.min(1, ratio)));
        return a * b * (t + Math.PI / 2 + Math.sin(2 * t) / 2) / 1;
      }
      case 'VERT_ELLIPSE': {
        const a2 = D / 2;
        const b2 = (geom2 || D * 0.67) / 2;
        if (d >= D) return Math.PI * a2 * b2;
        return this._getArea('CIRCULAR', D, 0, d) * (b2 / (D / 2)); // approximate
      }
      case 'EGG': {
        // Approximate as 3/2 × circular
        return this.getArea('CIRCULAR', D, 0, d) * 1.0;
      }
      case 'HORSESHOE': {
        return this.getArea('CIRCULAR', D, 0, d) * 1.05;
      }
      default: {
        // Default to circular approximation
        if (d <= 0) return 0;
        if (d >= D) return Math.PI * D * D / 4;
        const r = D / 2;
        const ratio = Math.max(-1, Math.min(1, 1 - d / r));
        const theta = 2 * Math.acos(ratio);
        return r * r / 2 * (theta - Math.sin(theta));
      }
    }
  }

  getWettedPerimeter(shape, geom1, geom2, depth) {
    const d = Math.max(0, Math.min(depth, geom1 || 1));
    const D = geom1 || 1;

    switch (shape) {
      case 'CIRCULAR': {
        if (d <= 0) return 0;
        if (d >= D) return Math.PI * D;
        const r = D / 2;
        const theta = 2 * Math.acos(Math.max(-1, Math.min(1, 1 - d / r)));
        return r * theta;
      }
      case 'RECT_CLOSED':
      case 'RECT_OPEN': {
        const w = geom2 || D;
        if (shape === 'RECT_OPEN') return w + 2 * d;
        return w + 2 * d; // closed same for wetted
      }
      default: {
        const r = D / 2;
        if (d <= 0) return 0;
        if (d >= D) return Math.PI * D;
        const theta = 2 * Math.acos(Math.max(-1, Math.min(1, 1 - d / r)));
        return r * theta;
      }
    }
  }

  getTopWidth(shape, geom1, geom2, depth) {
    const d = Math.max(0.0001, Math.min(depth, (geom1 || 1) - 0.0001));
    const D = geom1 || 1;

    switch (shape) {
      case 'CIRCULAR': {
        const r = D / 2;
        return 2 * Math.sqrt(Math.max(0, r * r - (r - d) * (r - d)));
      }
      case 'RECT_CLOSED':
      case 'RECT_OPEN':
        return geom2 || D;
      case 'TRAPEZOIDAL':
        return (geom2 || D * 0.5) + 2 * (geom2 || 1) * d;
      case 'TRIANGULAR':
        return 2 * (geom2 || 1) * d;
      default: {
        const r = D / 2;
        return 2 * Math.sqrt(Math.max(0, r * r - (r - d) * (r - d)));
      }
    }
  }

  getHydRadius(shape, geom1, geom2, depth) {
    const A = this.getArea(shape, geom1, geom2, depth);
    const P = this.getWettedPerimeter(shape, geom1, geom2, depth);
    return P > 0 ? A / P : 0;
  }

  _getAreaFull(shape, geom1, geom2) {
    return this.getArea(shape, geom1, geom2, geom1);
  }

  _getHydRadiusFull(shape, geom1, geom2) {
    return this.getHydRadius(shape, geom1, geom2, geom1);
  }

  // ═══════════════════════════════════════════════════
  // COMPUTED LINK DIAGNOSTICS
  // ═══════════════════════════════════════════════════

  getLinkDiagnostics(linkId, timestep) {
    const cd = this._conduitData.get(linkId);
    if (!cd) return null;

    const flow = this.L1.getLinkResult(linkId, 'flow', timestep);
    const depth = this.L1.getLinkResult(linkId, 'depth', timestep);
    const velocity = this.L1.getLinkResult(linkId, 'velocity', timestep);
    const capacity = this.L1.getLinkResult(linkId, 'capacity', timestep);

    const h1 = this.L1.getNodeResult(cd.fromNode, 'head', timestep);
    const h2 = this.L1.getNodeResult(cd.toNode, 'head', timestep);
    const y1 = h1 - cd.invertUp;
    const y2 = h2 - cd.invertDn;

    // Cross-section properties at midpoint
    const dMid = Math.max(0.001, Math.min(depth, cd.geom1));
    const aMid = this.getArea(cd.shape, cd.geom1, cd.geom2, dMid);
    const rMid = this.getHydRadius(cd.shape, cd.geom1, cd.geom2, dMid);
    const wMid = this.getTopWidth(cd.shape, cd.geom1, cd.geom2, dMid);

    // Upstream/downstream areas
    const a1 = this.getArea(cd.shape, cd.geom1, cd.geom2, Math.max(0, y1));
    const a2 = this.getArea(cd.shape, cd.geom1, cd.geom2, Math.max(0, y2));
    const r1 = this.getHydRadius(cd.shape, cd.geom1, cd.geom2, Math.max(0, y1));
    const r2 = this.getHydRadius(cd.shape, cd.geom1, cd.geom2, Math.max(0, y2));
    const w1 = this.getTopWidth(cd.shape, cd.geom1, cd.geom2, Math.max(0, y1));
    const w2 = this.getTopWidth(cd.shape, cd.geom1, cd.geom2, Math.max(0, y2));

    // Velocities at ends
    const v1 = a1 > 0.001 ? flow / a1 : 0;
    const v2 = a2 > 0.001 ? flow / a2 : 0;
    const vMid = aMid > 0.001 ? flow / aMid : velocity;

    // Froude number
    const celerity = wMid > 0 ? Math.sqrt(this.g * aMid / wMid) : 1;
    const froude = celerity > 0 ? Math.abs(vMid) / celerity : 0;

    // Flow classification
    let flowClass = 'UNKNOWN';
    if (depth <= 0.001) flowClass = 'DRY';
    else if (y1 <= 0.001 && y2 > 0.001) flowClass = 'UP_DRY';
    else if (y1 > 0.001 && y2 <= 0.001) flowClass = 'DN_DRY';
    else if (depth >= cd.geom1 * 0.97) flowClass = 'FULL';
    else if (froude < 0.95) flowClass = 'SUBCRITICAL';
    else if (froude > 1.05) flowClass = 'SUPERCRITICAL';
    else flowClass = 'CRITICAL';

    // Friction slope (Manning's equation inverted)
    let frictionSlope = 0;
    if (aMid > 0 && rMid > 0) {
      const qn = cd.n * Math.abs(flow) / (this.phi * aMid * Math.pow(rMid, 2 / 3));
      frictionSlope = qn * qn;
    }

    // Normal flow capacity at current depth
    const qNormal = cd.slope > 0 && rMid > 0 ?
      (this.phi / cd.n) * aMid * Math.pow(rMid, 2 / 3) * Math.sqrt(cd.slope) : 0;

    // Head losses
    const frictionLoss = frictionSlope * cd.length;
    const entryLoss = cd.entryLoss * v1 * v1 / (2 * this.g);
    const exitLoss = cd.exitLoss * v2 * v2 / (2 * this.g);
    const avgLossHead = cd.avgLoss * vMid * vMid / (2 * this.g) * cd.length;
    const totalHeadLoss = frictionLoss + entryLoss + exitLoss + avgLossHead;

    // Velocity heads
    const vhUp = v1 * v1 / (2 * this.g);
    const vhMid = vMid * vMid / (2 * this.g);
    const vhDn = v2 * v2 / (2 * this.g);

    // Energy Grade Line
    const eglUp = h1 + vhUp;
    const eglDn = h2 + vhDn;

    // Bernoulli energy balance check
    const bernoulliLHS = h1 + vhUp + entryLoss;
    const bernoulliRHS = h2 + vhDn + exitLoss + frictionLoss;
    const bernoulliError = bernoulliLHS - bernoulliRHS;
    const bernoulliPct = bernoulliLHS > 0 ? Math.abs(bernoulliError / bernoulliLHS * 100) : 0;

    // Hydraulic gradient
    const hydraulicGradient = cd.length > 0 ? (h1 - h2) / cd.length : 0;

    // Specific energy
    const specificEnergy = depth + vMid * vMid / (2 * this.g);

    // Critical depth (approximate for circular)
    const criticalDepth = this._getCriticalDepth(cd, Math.abs(flow));

    // Self-cleansing check
    const selfCleansing = Math.abs(vMid) >= 2.0; // 2 ft/s minimum
    const erosionRisk = Math.abs(vMid) > 15.0;

    return {
      // Input from Level 1
      flow, depth, velocity, capacity,
      h1, h2, y1, y2,

      // Cross-section at midpoint
      aMid, rMid, wMid,

      // Cross-section at ends
      a1, a2, r1, r2, w1, w2, v1, v2,

      // Hydraulic characterization
      froude,
      flowClass,
      frictionSlope,
      hydraulicGradient,
      qNormal,
      qFull: cd.qFull,
      specificEnergy,
      criticalDepth,

      // Head losses
      frictionLoss,
      entryLoss,
      exitLoss,
      avgLossHead,
      totalHeadLoss,

      // Velocity heads
      vhUp, vhMid, vhDn,

      // Energy grade line
      hglUp: h1, hglDn: h2,
      eglUp, eglDn,

      // Bernoulli balance
      bernoulliLHS, bernoulliRHS,
      bernoulliError, bernoulliPct,

      // Engineering checks
      selfCleansing,
      erosionRisk,
      capacityUsed: capacity,

      // Pipe properties
      slope: cd.slope,
      slopePercent: cd.slope * 100,
      length: cd.length,
      diameter: cd.geom1,
      roughness: cd.n,
      shape: cd.shape,
    };
  }

  _getCriticalDepth(cd, Q) {
    if (Q <= 0 || !cd) return 0;
    // Newton-Raphson for critical depth where Fr = 1
    // Q² / g = A³ / T  →  solve for d
    const target = Q * Q / this.g;
    let d = cd.geom1 * 0.5; // initial guess
    for (let iter = 0; iter < 20; iter++) {
      const A = this.getArea(cd.shape, cd.geom1, cd.geom2, d);
      const T = this.getTopWidth(cd.shape, cd.geom1, cd.geom2, d);
      if (T <= 0) break;
      const f = A * A * A / T - target;
      const dA = T; // dA/dd ≈ T
      const dT = 0; // approximate
      const df = (3 * A * A * T * dA - A * A * A * dT) / (T * T);
      if (Math.abs(df) < 1e-10) break;
      const dd = -f / df;
      d = Math.max(0.001, Math.min(cd.geom1 * 0.99, d + dd));
      if (Math.abs(dd) < 0.0001) break;
    }
    return d;
  }

  // ═══════════════════════════════════════════════════
  // COMPUTED NODE DIAGNOSTICS
  // ═══════════════════════════════════════════════════

  getNodeDiagnostics(nodeId, timestep) {
    const depth = this.L1.getNodeResult(nodeId, 'depth', timestep);
    const head = this.L1.getNodeResult(nodeId, 'head', timestep);
    const volume = this.L1.getNodeResult(nodeId, 'volume', timestep);
    const latInflow = this.L1.getNodeResult(nodeId, 'latInflow', timestep);
    const totalInflow = this.L1.getNodeResult(nodeId, 'totalInflow', timestep);
    const flooding = this.L1.getNodeResult(nodeId, 'flooding', timestep);

    const invert = this.L1.nodeInverts[nodeId] || 0;
    const maxDepth = this.L1.nodeMaxDepths[nodeId] || 0;
    const nodeType = NODE_TYPES[this.L1.nodeTypes[nodeId]] || 'junction';

    // Surcharge detection
    const crownElev = this._getNodeCrown(nodeId);
    const isSurcharged = head > crownElev && crownElev > 0;
    const surchargeDepth = isSurcharged ? head - crownElev : 0;

    // Depth ratio
    const depthRatio = maxDepth > 0 ? depth / maxDepth : 0;

    // Freeboard
    const freeboard = maxDepth > 0 ? maxDepth - depth : 0;

    // Net inflow (inflow - outflow estimated from continuity)
    const linkInflow = totalInflow - latInflow;

    // Estimated outflow (total inflow - flooding - storage change)
    const prevVolume = timestep > 0 ? this.L1.getNodeResult(nodeId, 'volume', timestep - 1) : volume;
    const dVdt = (volume - prevVolume) / this.L1.reportStep;
    const estimatedOutflow = totalInflow - flooding - dVdt;

    // Connected links info
    const upstreamLinks = [];
    const downstreamLinks = [];
    if (this.model) {
      for (const [lid, c] of this.model.conduits) {
        if (c.toNode === nodeId) upstreamLinks.push(lid);
        if (c.fromNode === nodeId) downstreamLinks.push(lid);
      }
      for (const [lid, p] of this.model.pumps) {
        if (p.toNode === nodeId) upstreamLinks.push(lid);
        if (p.fromNode === nodeId) downstreamLinks.push(lid);
      }
    }

    return {
      depth, head, volume, latInflow, totalInflow, flooding,
      invert, maxDepth, nodeType,
      crownElev,
      isSurcharged, surchargeDepth,
      depthRatio, freeboard,
      linkInflow,
      estimatedOutflow,
      dVdt,
      upstreamLinks, downstreamLinks,
      degree: upstreamLinks.length + downstreamLinks.length,
    };
  }

  _getNodeCrown(nodeId) {
    if (!this.model) return 0;
    let maxCrown = 0;
    const invert = this.L1.nodeInverts[nodeId] || 0;
    for (const [, cd] of this._conduitData) {
      if (cd.fromNode === nodeId) {
        const crown = cd.invertUp + cd.geom1;
        if (crown > maxCrown) maxCrown = crown;
      }
      if (cd.toNode === nodeId) {
        const crown = cd.invertDn + cd.geom1;
        if (crown > maxCrown) maxCrown = crown;
      }
    }
    return maxCrown || (invert + (this.L1.nodeMaxDepths[nodeId] || 0));
  }

  // ═══════════════════════════════════════════════════
  // SYSTEM-WIDE DIAGNOSTICS
  // ═══════════════════════════════════════════════════

  getSystemDiagnostics(timestep) {
    const sys = {};
    for (const v of SYSTEM_VARS) {
      sys[v] = this.L1.getSystemResult(v, timestep);
    }

    // Mass balance check
    const totalIn = sys.totalLatInflow || 0;
    const totalOut = (sys.outflow || 0) + (sys.flooding || 0) + (sys.evaporation || 0) + (sys.seepageLoss || 0);
    const storage = sys.storageVol || 0;
    const prevStorage = timestep > 0 ? this.L1.getSystemResult('storageVol', timestep - 1) : storage;
    const dStorage = storage - prevStorage;
    const massBalanceError = totalIn > 0 ? (totalIn - totalOut - dStorage / this.L1.reportStep) / totalIn * 100 : 0;

    // Count flooding nodes
    let floodingNodeCount = 0;
    let maxFloodingNode = '';
    let maxFloodingRate = 0;
    for (const name of this.L1.nodeNames) {
      const fl = this.L1.getNodeResult(name, 'flooding', timestep);
      if (fl > 0.001) {
        floodingNodeCount++;
        if (fl > maxFloodingRate) {
          maxFloodingRate = fl;
          maxFloodingNode = name;
        }
      }
    }

    // Count surcharged links
    let surchargedCount = 0;
    let fullCount = 0;
    for (const name of this.L1.linkNames) {
      const cap = this.L1.getLinkResult(name, 'capacity', timestep);
      if (cap > 1.0) surchargedCount++;
      if (cap > 0.95) fullCount++;
    }

    // Velocity extremes
    let maxVelocity = 0, maxVelLink = '';
    let minVelocity = Infinity, minVelLink = '';
    for (const name of this.L1.linkNames) {
      const v = Math.abs(this.L1.getLinkResult(name, 'velocity', timestep));
      const f = this.L1.getLinkResult(name, 'flow', timestep);
      if (v > maxVelocity && Math.abs(f) > 0.001) { maxVelocity = v; maxVelLink = name; }
      if (v < minVelocity && Math.abs(f) > 0.01) { minVelocity = v; minVelLink = name; }
    }

    return {
      ...sys,
      massBalanceError,
      floodingNodeCount,
      maxFloodingNode, maxFloodingRate,
      surchargedCount, fullCount,
      maxVelocity, maxVelLink,
      minVelocity: minVelocity === Infinity ? 0 : minVelocity, minVelLink,
    };
  }

  // ═══════════════════════════════════════════════════
  // FULL SYSTEM SUMMARY (all timesteps)
  // ═══════════════════════════════════════════════════

  getFullSummary() {
    const nodeSummary = {};
    for (const name of this.L1.nodeNames) {
      const s = { maxDepth: 0, maxHead: 0, maxInflow: 0, maxFlooding: 0,
                  hoursFlooded: 0, totalFloodVol: 0, timeOfMaxDepth: 0 };
      for (let t = 0; t < this.L1.nPeriods; t++) {
        const d = this.L1.getNodeResult(name, 'depth', t);
        const h = this.L1.getNodeResult(name, 'head', t);
        const inf = this.L1.getNodeResult(name, 'totalInflow', t);
        const fl = this.L1.getNodeResult(name, 'flooding', t);
        if (d > s.maxDepth) { s.maxDepth = d; s.timeOfMaxDepth = t; }
        if (h > s.maxHead) s.maxHead = h;
        if (inf > s.maxInflow) s.maxInflow = inf;
        if (fl > s.maxFlooding) s.maxFlooding = fl;
        if (fl > 0.001) {
          s.hoursFlooded += this.L1.reportStep / 3600;
          s.totalFloodVol += fl * this.L1.reportStep;
        }
      }
      s.timeOfMaxDepthStr = this.L1._formatTime(s.timeOfMaxDepth);
      nodeSummary[name] = s;
    }

    const linkSummary = {};
    for (const name of this.L1.linkNames) {
      const s = { maxFlow: 0, maxVelocity: 0, maxDepth: 0, maxCapacity: 0,
                  hoursFull: 0, timeOfMaxFlow: 0 };
      const cd = this._conduitData.get(name);
      for (let t = 0; t < this.L1.nPeriods; t++) {
        const f = Math.abs(this.L1.getLinkResult(name, 'flow', t));
        const v = Math.abs(this.L1.getLinkResult(name, 'velocity', t));
        const d = this.L1.getLinkResult(name, 'depth', t);
        const c = this.L1.getLinkResult(name, 'capacity', t);
        if (f > s.maxFlow) { s.maxFlow = f; s.timeOfMaxFlow = t; }
        if (v > s.maxVelocity) s.maxVelocity = v;
        if (d > s.maxDepth) s.maxDepth = d;
        if (c > s.maxCapacity) s.maxCapacity = c;
        if (c >= 0.95) s.hoursFull += this.L1.reportStep / 3600;
      }
      s.timeOfMaxFlowStr = this.L1._formatTime(s.timeOfMaxFlow);
      if (cd) {
        s.slope = cd.slope;
        s.slopePercent = cd.slope * 100;
        s.qFull = cd.qFull;
        s.pctCapacityUsed = cd.qFull > 0 ? s.maxFlow / cd.qFull * 100 : 0;
      }
      linkSummary[name] = s;
    }

    return { nodeSummary, linkSummary };
  }

/**
 * LEVEL 3 — Engine Internal Variables
 *
 * Provides access to solver internals that are NOT in the standard .out file.
 * Two modes:
 *   A) ESTIMATED: Reconstruct solver internals from Level 1+2 data using the
 *      same equations the engine uses. ~85% accurate for well-behaved systems.
 *   B) EXTENDED: Read directly from a modified .out file that has
 *      MAX_NODE_RESULTS/MAX_LINK_RESULTS expanded to 100 (RED build).
 *
 * Variables reconstructed:
 *   NODE: dQ/dH (Jacobian), F(H) residual, NR denominator, convergence estimate
 *   LINK: DQ1-DQ6 momentum terms, σ damping, friction slope, Bernoulli balance
 *   SUBCATCH: Infiltration state (GA/Horton/CN), runoff breakdown, LID layers
 */

export class SWMMOutputLevel3 {
  constructor(level1, level2, model) {
    this.L1 = level1;
    this.L2 = level2;
    this.model = model;

    // Check if this is an extended .out file (RED build)
    this.isExtended = this._detectExtendedOutput();
  }

  _detectExtendedOutput() {
    // Extended output has nNodeVars > 6 or nLinkVars > 5
    return (this.L1.nNodeVars > 10 || this.L1.nLinkVars > 10);
  }

  // ═══════════════════════════════════════════════════
  // LINK MOMENTUM EQUATION TERMS (DQ1-DQ6)
  // ═══════════════════════════════════════════════════

  getLinkMomentum(linkId, timestep) {
    const diag = this.L2.getLinkDiagnostics(linkId, timestep);
    if (!diag) return null;

    const cd = this.L2._conduitData.get(linkId);
    if (!cd) return null;

    const dt = this.L1.reportStep;
    const g = this.L2.g;
    const phi = this.L2.phi;

    // Previous timestep flow
    const qOld = timestep > 0 ? this.L1.getLinkResult(linkId, 'flow', timestep - 1) : 0;
    const qNew = diag.flow;

    // σ (inertial damping factor)
    // Estimate: σ = 1 for free surface, transitions to 0 when surcharged
    const dOverD = diag.depth / cd.geom1;
    let sigma;
    if (dOverD < 0.9) sigma = 1.0;
    else if (dOverD > 1.1) sigma = 0.0;
    else sigma = Math.max(0, (1.1 - dOverD) / 0.2); // linear transition

    // ─── DQ1: Inertia ───
    const DQ1 = sigma * qOld;

    // ─── DQ2: Pressure Gradient ───
    const DQ2 = sigma * dt * g * diag.aMid * (diag.h1 - diag.h2) / cd.length;

    // ─── DQ3: Friction ───
    // Sf from Manning: Sf = (n|Q| / (φ·A·R^⅔))²
    const DQ3 = dt * g * diag.aMid * diag.frictionSlope * Math.sign(qNew);

    // ─── DQ4: Entry/Exit/Average Losses ───
    const DQ4 = diag.entryLoss + diag.exitLoss + diag.avgLossHead;

    // ─── DQ5: Lateral Inflow Momentum ───
    // Approximate from node lateral inflows
    const latInflowUS = this.L1.getNodeResult(cd.fromNode, 'latInflow', timestep);
    const latInflowDS = this.L1.getNodeResult(cd.toNode, 'latInflow', timestep);
    const DQ5 = 0.5 * dt * (latInflowUS + latInflowDS) * diag.velocity / cd.length;

    // ─── DQ6: Convective Acceleration ───
    // ∂(βQ²/A)/∂x ≈ β(Q²/A2 - Q²/A1)/L
    const beta = 1.0;
    const convA1 = diag.a1 > 0.001 ? beta * qNew * qNew / diag.a1 : 0;
    const convA2 = diag.a2 > 0.001 ? beta * qNew * qNew / diag.a2 : 0;
    const DQ6 = sigma * dt * (convA2 - convA1) / cd.length;

    // ─── Momentum Balance ───
    const drivingForces = DQ1 + DQ2 + DQ5;
    const resistingForces = DQ3 + DQ4 + DQ6;
    const netForce = drivingForces - resistingForces;

    // Denominator in momentum equation
    const sfDenom = diag.frictionSlope > 0 && diag.aMid > 0 ?
      dt * g * cd.n * cd.n * Math.abs(qNew) / (phi * phi * diag.aMid * Math.pow(diag.rMid, 4/3)) : 0;
    const denominator = 1.0 + sfDenom;

    // Reconstructed Q from momentum equation
    const qReconstructed = denominator > 0 ? (DQ1 + DQ2 + DQ5) / denominator : 0;

    // Momentum equation residual
    const momentumResidual = qNew - qReconstructed;
    const momentumBalancePct = Math.abs(qNew) > 0.001 ?
      Math.abs(momentumResidual / qNew * 100) : 0;

    // ─── Relative magnitudes ───
    const totalMagnitude = Math.abs(DQ1) + Math.abs(DQ2) + Math.abs(DQ3) +
                           Math.abs(DQ4) + Math.abs(DQ5) + Math.abs(DQ6);
    const pctInertia = totalMagnitude > 0 ? Math.abs(DQ1) / totalMagnitude * 100 : 0;
    const pctPressure = totalMagnitude > 0 ? Math.abs(DQ2) / totalMagnitude * 100 : 0;
    const pctFriction = totalMagnitude > 0 ? Math.abs(DQ3) / totalMagnitude * 100 : 0;
    const pctMinorLoss = totalMagnitude > 0 ? Math.abs(DQ4) / totalMagnitude * 100 : 0;
    const pctLateral = totalMagnitude > 0 ? Math.abs(DQ5) / totalMagnitude * 100 : 0;
    const pctConvective = totalMagnitude > 0 ? Math.abs(DQ6) / totalMagnitude * 100 : 0;

    // ─── Courant number ───
    const waveCelerity = diag.wMid > 0 ? Math.sqrt(g * diag.aMid / diag.wMid) : 0;
    const courant = cd.length > 0 ? (Math.abs(diag.velocity) + waveCelerity) * dt / cd.length : 0;

    return {
      // Momentum terms
      DQ1, DQ2, DQ3, DQ4, DQ5, DQ6,
      sigma,

      // Balance
      drivingForces, resistingForces, netForce,
      denominator,
      qOld, qNew, qReconstructed,
      momentumResidual, momentumBalancePct,

      // Relative magnitudes (%)
      pctInertia, pctPressure, pctFriction,
      pctMinorLoss, pctLateral, pctConvective,

      // Stability metrics
      courant,
      waveCelerity,

      // All Level 2 diagnostics included
      ...diag,
    };
  }

  // ═══════════════════════════════════════════════════
  // NODE SOLVER INTERNALS (Estimated)
  // ═══════════════════════════════════════════════════

  getNodeSolverState(nodeId, timestep) {
    const nd = this.L2.getNodeDiagnostics(nodeId, timestep);
    if (!nd) return null;

    const dt = this.L1.reportStep;

    // Surface area estimate
    // For junctions: use MIN_SURFAREA (default 12.566 ft² = π × 2²)
    const minSurfArea = 12.566;
    let surfArea = minSurfArea;
    if (nd.nodeType === 'storage' && this.model) {
      const stor = this.model.storage.get(nodeId);
      if (stor && stor.storageShape === 'FUNCTIONAL') {
        // A = a × d^b + c
        surfArea = (stor.storageCoeffA || 1000) * Math.pow(Math.max(0.01, nd.depth), stor.storageExponentB || 0) + (stor.storageConstantC || 0);
      }
    }

    // ─── Estimated dQ/dH (Jacobian) ───
    // dQ/dH ≈ Σ(∂Q_link/∂H_node) for all connected links
    let dqdh = 0;
    for (const lid of [...nd.upstreamLinks, ...nd.downstreamLinks]) {
      const linkDiag = this.L2.getLinkDiagnostics(lid, timestep);
      if (!linkDiag) continue;
      const cd = this.L2._conduitData.get(lid);
      if (!cd || cd.length <= 0) continue;

      // ∂Q/∂H ≈ g × A_mid × Δt / L (from pressure gradient sensitivity)
      const sensitivity = this.L2.g * linkDiag.aMid * dt / cd.length;
      dqdh += sensitivity;
    }

    // ─── NR Denominator ───
    const nrDenom = dqdh + surfArea / dt;

    // ─── F(H) Residual ───
    // F = ΣQ_in - ΣQ_out - dV/dt
    const fResidual = nd.dVdt > 0 ? nd.totalInflow - nd.estimatedOutflow - nd.dVdt : 0;

    // ─── Head Correction ───
    const headCorrection = nrDenom > 0 ? -fResidual / nrDenom : 0;

    // ─── Convergence estimate ───
    const isConverged = Math.abs(headCorrection) < 0.005; // HEAD_TOLERANCE default
    const convergenceQuality = Math.abs(headCorrection) < 0.001 ? 'EXCELLENT' :
                               Math.abs(headCorrection) < 0.005 ? 'GOOD' :
                               Math.abs(headCorrection) < 0.05 ? 'MARGINAL' : 'POOR';

    // ─── Stiffness indicator ───
    // When dQ/dH is small relative to surfArea/dt, the system is storage-dominated (stiff)
    const stiffnessRatio = surfArea / dt > 0 ? dqdh / (surfArea / dt) : 0;
    const isStiff = stiffnessRatio < 0.1;

    return {
      ...nd,
      surfArea,
      dqdh,
      nrDenom,
      fResidual,
      headCorrection,
      isConverged,
      convergenceQuality,
      stiffnessRatio,
      isStiff,
    };
  }

  // ═══════════════════════════════════════════════════
  // SUBCATCHMENT INFILTRATION STATE (Estimated)
  // ═══════════════════════════════════════════════════

  getSubcatchInfiltration(subcatchId, timestep) {
    const rainfall = this.L1.getSubcatchResult(subcatchId, 'rainfall', timestep);
    const infil = this.L1.getSubcatchResult(subcatchId, 'infil', timestep);
    const runoff = this.L1.getSubcatchResult(subcatchId, 'runoff', timestep);
    const evap = this.L1.getSubcatchResult(subcatchId, 'evap', timestep);
    const soilMoisture = this.L1.getSubcatchResult(subcatchId, 'soilMoisture', timestep);

    const sub = this.model?.subcatchments.get(subcatchId);
    if (!sub) return { rainfall, infil, runoff, evap, soilMoisture };

    const pctImperv = sub.pctImperv || 0;
    const area = sub.area || 0;

    // Infiltration method detection
    const method = (this.model.options?.INFILTRATION || 'GREEN_AMPT').toUpperCase();

    // Estimate cumulative infiltration (integrate from start)
    let cumInfil = 0;
    for (let t = 0; t <= timestep; t++) {
      cumInfil += this.L1.getSubcatchResult(subcatchId, 'infil', t) * this.L1.reportStep / 3600;
    }

    // Estimate cumulative rainfall
    let cumRain = 0;
    for (let t = 0; t <= timestep; t++) {
      cumRain += this.L1.getSubcatchResult(subcatchId, 'rainfall', t) * this.L1.reportStep / 3600;
    }

    // Estimate cumulative runoff
    let cumRunoff = 0;
    for (let t = 0; t <= timestep; t++) {
      cumRunoff += this.L1.getSubcatchResult(subcatchId, 'runoff', t) * this.L1.reportStep / 3600;
    }

    // Runoff coefficient
    const runoffCoeff = cumRain > 0 ? cumRunoff / cumRain : 0;

    // Green-Ampt estimates
    let gaState = null;
    if (method.includes('GREEN_AMPT') && sub.conductivity) {
      const Ks = sub.conductivity; // in/hr
      const psi = sub.suctionHead || 0; // in
      const IMD = sub.initDeficit || 0;

      // Current infiltration capacity: f = Ks × (1 + ψ×Δθ/F)
      const fCapacity = cumInfil > 0.01 ?
        Ks * (1 + psi * IMD / cumInfil) : Ks * 10; // high when F is small

      // Wetting front depth
      const wettingFront = IMD > 0 ? cumInfil / IMD : cumInfil;

      // Time to saturation estimate
      const excessRate = rainfall - fCapacity;
      const isSaturated = excessRate > 0 && cumInfil > psi * IMD;

      gaState = {
        method: 'GREEN_AMPT',
        Ks, psi, IMD,
        cumulativeF: cumInfil,
        currentCapacity: fCapacity,
        currentRate: infil,
        wettingFront,
        isSaturated,
        excessRate: Math.max(0, excessRate),
      };
    }

    // Horton estimates
    let hortonState = null;
    if (method.includes('HORTON') && sub.conductivity !== undefined) {
      // Not enough info to fully reconstruct, but can estimate
      hortonState = {
        method: 'HORTON',
        currentRate: infil,
        cumulativeF: cumInfil,
      };
    }

    // Runoff breakdown estimate
    const impervRunoff = rainfall > 0 ? runoff * (pctImperv / 100) * 1.5 : 0; // approximate
    const pervRunoff = runoff - impervRunoff;

    return {
      rainfall, infil, runoff, evap, soilMoisture,
      cumRain, cumInfil, cumRunoff,
      runoffCoeff,
      pctImperv,
      area,

      // Estimated breakdown
      impervRunoff: Math.max(0, impervRunoff),
      pervRunoff: Math.max(0, pervRunoff),

      // Infiltration state
      infiltrationMethod: method,
      gaState,
      hortonState,

      // Ponding check
      isPonding: rainfall > infil && rainfall > 0,
      excessRainfall: Math.max(0, rainfall - infil),
    };
  }

  // ═══════════════════════════════════════════════════
  // EXTENDED .OUT FILE READER (RED/Innovyze builds)
  // ═══════════════════════════════════════════════════

  getExtendedNodeResult(nodeId, extVarIdx, timestep) {
    if (!this.isExtended) return null;
    // Extended variables start at index 6 (after the standard 6)
    return this.L1.getNodeResult(nodeId, extVarIdx, timestep);
  }

  getExtendedLinkResult(linkId, extVarIdx, timestep) {
    if (!this.isExtended) return null;
    return this.L1.getLinkResult(linkId, extVarIdx, timestep);
  }

  getExtendedSubcatchResult(subcatchId, extVarIdx, timestep) {
    if (!this.isExtended) return null;
    return this.L1.getSubcatchResult(subcatchId, extVarIdx, timestep);
  }

  // ═══════════════════════════════════════════════════
  // SYSTEM SOLVER DIAGNOSTICS
  // ═══════════════════════════════════════════════════

  getSystemSolverState(timestep) {
    const sysDiag = this.L2.getSystemDiagnostics(timestep);

    // Compute per-link Courant numbers
    const courantNumbers = [];
    for (const name of this.L1.linkNames) {
      const mom = this.getLinkMomentum(name, timestep);
      if (mom && mom.courant > 0) {
        courantNumbers.push({ link: name, courant: mom.courant });
      }
    }
    courantNumbers.sort((a, b) => b.courant - a.courant);

    // Compute per-node convergence quality
    const nodeConvergence = [];
    for (const name of this.L1.nodeNames) {
      const ns = this.getNodeSolverState(name, timestep);
      if (ns) {
        nodeConvergence.push({
          node: name,
          quality: ns.convergenceQuality,
          headCorrection: ns.headCorrection,
          dqdh: ns.dqdh,
          isStiff: ns.isStiff,
        });
      }
    }

    // Find worst convergence nodes
    const worstNodes = nodeConvergence
      .sort((a, b) => Math.abs(b.headCorrection) - Math.abs(a.headCorrection))
      .slice(0, 10);

    // Find highest Courant links
    const worstCourant = courantNumbers.slice(0, 10);

    // Overall stability assessment
    const maxCourant = courantNumbers.length > 0 ? courantNumbers[0].courant : 0;
    const maxCorrection = worstNodes.length > 0 ? Math.abs(worstNodes[0].headCorrection) : 0;
    const poorNodes = nodeConvergence.filter(n => n.quality === 'POOR').length;

    let stabilityRating;
    if (maxCourant < 1 && maxCorrection < 0.005 && poorNodes === 0) stabilityRating = 'EXCELLENT';
    else if (maxCourant < 2 && maxCorrection < 0.05 && poorNodes < 3) stabilityRating = 'GOOD';
    else if (maxCourant < 5 && poorNodes < 10) stabilityRating = 'MARGINAL';
    else stabilityRating = 'POOR';

    return {
      ...sysDiag,
      maxCourant,
      maxCorrection,
      poorConvergenceNodes: poorNodes,
      stabilityRating,
      worstNodes,
      worstCourant,
      totalNodes: this.L1.nNodes,
      totalLinks: this.L1.nLinks,
    };
  }
}

        highVelocityLinks: highVelocityLinks.length,
        adverseSlopeLinks: adverseSlopeLinks.length,

    };
  }

## What Each Level Provides

### Level 1 — Standard EPA Output (34 variables)
- **6 node variables**: depth, head, volume, lateral inflow, total inflow, flooding
- **5 link variables**: flow, depth, velocity, volume, capacity (d/D)
- **8 subcatchment variables**: rainfall, snow, evap, infiltration, runoff, GW flow, GW elevation, soil moisture
- **15 system variables**: temperature, rainfall, snow, evap, runoff, DWF, GW, RDII, inflows, flooding, outflow, storage, evaporation, seepage
- **Full time series extraction** for any variable at any object
- **Statistics**: min, max, mean, time of max for every variable
- **Legend ranges** for map animation coloring
- **Object properties from header**: node types/inverts/max depths, link types/offsets

### Level 2 — Hydraulic Diagnostics (~40 derived variables)
Computed from Level 1 + model geometry without requiring engine modifications:
- **Cross-section geometry** for 10+ shapes: area, wetted perimeter, hydraulic radius, top width at any depth
- **Froude number** and **flow classification** (DRY, SUBCRITICAL, SUPERCRITICAL, CRITICAL, FULL)
- **Friction slope** from Manning's equation inversion
- **Normal flow capacity** and slope from pipe geometry
- **Head losses**: friction, entry, exit, average — all separately computed
- **Energy Grade Line**: HGL and EGL at both pipe ends
- **Bernoulli energy balance check**: LHS vs RHS with error percentage
- **Critical depth** via Newton-Raphson iteration
- **Velocity heads** at upstream, midpoint, and downstream
- **Node diagnostics**: surcharge detection, freeboard, depth ratio, crown elevation, estimated outflow
- **System diagnostics**: flooding node count, surcharged link count, velocity extremes, mass balance error
- **Full simulation summary**: max/min statistics for every node and link

### Level 3 — Solver Internals (~100+ variables)
Estimated from Levels 1+2 or read from extended output files:
- **DQ1–DQ6 momentum equation terms**: inertia, pressure gradient, friction, minor losses, lateral momentum, convective acceleration — each computed separately
- **σ (inertial damping factor)**: estimated from depth/diameter ratio
- **Momentum balance**: driving forces vs resisting forces, net force, reconstructed Q
- **Relative term magnitudes**: what percentage of the momentum equation is friction vs pressure vs inertia
- **Courant number** per link per timestep
- **Node solver state**: estimated dQ/dH Jacobian, NR denominator, F(H) residual, head correction, convergence quality
- **Stiffness detection**: ratio of Jacobian to storage term
- **Subcatchment infiltration state**: cumulative F, current capacity, wetting front depth (Green-Ampt), ponding detection
- **System solver state**: stability rating, worst convergence nodes, highest Courant links
- **Comprehensive summary**: all problem objects identified and ranked by severity
