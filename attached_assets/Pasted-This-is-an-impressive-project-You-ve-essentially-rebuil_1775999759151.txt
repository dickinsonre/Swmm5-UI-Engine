```typescript
// client/src/lib/store/projectStore.ts
import { create } from 'zustand';
import { immer } from 'zustand/middleware/immer';

interface ProjectState {
  project: SwmmProject | null;
  results: SimulationResults | null;
  fileName: string;
  simStatus: 'none' | 'running' | 'current' | 'outdated';
  selectedObj: SelectedObject | null;
  interactionMode: string;
  
  // Actions
  setProject: (project: SwmmProject) => void;
  updateObject: (type: string, id: string, updates: Partial<any>) => void;
  runSimulation: (engine: SwmmEngine) => Promise<void>;
  selectObject: (obj: SelectedObject | null) => void;
}

export const useProjectStore = create<ProjectState>()(
  immer((set, get) => ({
    project: null,
    results: null,
    fileName: 'untitled.inp',
    simStatus: 'none',
    selectedObj: null,
    interactionMode: 'select',
    
    setProject: (project) => set({ project, simStatus: 'outdated' }),
    
    updateObject: (type, id, updates) => set((state) => {
      if (!state.project) return;
      const collection = state.project[type as keyof SwmmProject];
      if (Array.isArray(collection)) {
        const idx = collection.findIndex((o: any) => o.id === id);
        if (idx >= 0) Object.assign(collection[idx], updates);
      }
      state.simStatus = 'outdated';
    }),
    
    runSimulation: async (engine) => {
      const { project } = get();
      if (!project) return;
      set({ simStatus: 'running' });
      try {
        const results = await engine.run(project);
        set({ results, simStatus: 'current' });
      } catch (e) {
        set({ simStatus: 'none' });
        throw e;
      }
    }
  }))
);
```

```typescript
// client/src/components/swmm/NetworkMap/OptimizedNetworkMap.tsx
import { useRef, useEffect, useCallback, memo } from 'react';

interface NetworkMapProps {
  project: SwmmProject;
  results: SimulationResults | null;
  timeStep: number;
  nodeTheme: string;
  linkTheme: string;
  subcatchTheme: string;
}

export const OptimizedNetworkMap = memo(function NetworkMap({
  project, results, timeStep, nodeTheme, linkTheme, subcatchTheme
}: NetworkMapProps) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const bgCanvasRef = useRef<HTMLCanvasElement>(null);
  const requestRef = useRef<number>();
  
  // Pre-render static background (grid, subcatchments, links without results)
  useEffect(() => {
    const bgCanvas = bgCanvasRef.current;
    if (!bgCanvas) return;
    
    const ctx = bgCanvas.getContext('2d');
    if (!ctx) return;
    
    // Render grid
    renderGrid(ctx, project.mapExtent);
    
    // Render subcatchments (static geometry)
    renderSubcatchments(ctx, project.polygons, project.subcatchments);
    
    // Render links (static geometry - no flow arrows)
    renderLinksStatic(ctx, project.conduits, project.coordinates, project.vertices);
    
    // Render nodes (static geometry - no depth fills)
    renderNodesStatic(ctx, project.junctions, project.outfalls, project.coordinates);
    
  }, [project]); // Only re-render when project changes
  
  // Render dynamic overlay (results-dependent)
  const renderOverlay = useCallback(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    
    const ctx = canvas.getContext('2d');
    if (!ctx) return;
    
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    
    if (!results) return;
    
    const ts = results.timeSteps[timeStep];
    if (!ts) return;
    
    // Render depth fills
    renderDepthFills(ctx, project.junctions, ts.nodes, nodeTheme);
    
    // Render flooding halos
    renderFloodingHalos(ctx, project.junctions, ts.nodes);
    
    // Render flow arrows
    renderFlowArrows(ctx, project.conduits, ts.links, project.coordinates, project.vertices);
    
  }, [project, results, timeStep, nodeTheme]);
  
  useEffect(() => {
    renderOverlay();
  }, [renderOverlay]);
  
  // RAF-based pan/zoom for smooth interaction
  const handlePanZoom = useCallback((dx: number, dy: number, zoom: number) => {
    if (requestRef.current) cancelAnimationFrame(requestRef.current);
    
    requestRef.current = requestAnimationFrame(() => {
      // Apply transform to both canvases
      applyTransform(canvasRef.current, dx, dy, zoom);
      applyTransform(bgCanvasRef.current, dx, dy, zoom);
    });
  }, []);
  
  return (
    <div className="relative w-full h-full">
      <canvas 
        ref={bgCanvasRef}
        className="absolute inset-0"
        style={{ pointerEvents: 'none' }}
      />
      <canvas 
        ref={canvasRef}
        className="absolute inset-0"
      />
    </div>
  );
});
```

```typescript
// client/src/workers/swmmWorker.ts
import { parseSwmmOut } from '../lib/swmm-out-parser';
import { computeExtendedVariables } from '../lib/swmm-engine';

self.onmessage = async (e) => {
  const { type, payload } = e.data;
  
  switch (type) {
    case 'PARSE_OUT': {
      const { buffer, project } = payload;
      const results = parseSwmmOut(buffer, project);
      const extended = computeExtendedVariables(project, results);
      self.postMessage({ type: 'PARSE_COMPLETE', results: extended });
      break;
    }
    
    case 'COMPUTE_EXTENDED': {
      const { project, results } = payload;
      const extended = computeExtendedVariables(project, results);
      self.postMessage({ type: 'EXTENDED_COMPLETE', extended });
      break;
    }
  }
};

// client/src/hooks/useSwmmWorker.ts
export function useSwmmWorker() {
  const workerRef = useRef<Worker | null>(null);
  
  useEffect(() => {
    workerRef.current = new Worker(
      new URL('../workers/swmmWorker.ts', import.meta.url),
      { type: 'module' }
    );
    
    return () => workerRef.current?.terminate();
  }, []);
  
  const parseResults = useCallback((buffer: ArrayBuffer, project: SwmmProject) => {
    return new Promise<SimulationResults>((resolve) => {
      const handler = (e: MessageEvent) => {
        if (e.data.type === 'PARSE_COMPLETE') {
          workerRef.current?.removeEventListener('message', handler);
          resolve(e.data.results);
        }
      };
      
      workerRef.current?.addEventListener('message', handler);
      workerRef.current?.postMessage({
        type: 'PARSE_OUT',
        payload: { buffer, project }
      });
    });
  }, []);
  
  return { parseResults };
}
```

### Use Replit's Persistent Storage for Cache

```typescript
// client/src/lib/cache/replitCache.ts
const CACHE_KEY = 'swmm_cache_v1';

interface CacheEntry<T> {
  data: T;
  timestamp: number;
  ttl: number;
}

export class ReplitCache {
  private storage: Storage;
  
  constructor() {
    this.storage = localStorage; // Replit persists this across sessions
  }
  
  get<T>(key: string): T | null {
    const entry = this.storage.getItem(`${CACHE_KEY}:${key}`);
    if (!entry) return null;
    
    const parsed: CacheEntry<T> = JSON.parse(entry);
    if (Date.now() - parsed.timestamp > parsed.ttl) {
      this.storage.removeItem(`${CACHE_KEY}:${key}`);
      return null;
    }
    
    return parsed.data;
  }
  
  set<T>(key: string, data: T, ttlMinutes = 60): void {
    const entry: CacheEntry<T> = {
      data,
      timestamp: Date.now(),
      ttl: ttlMinutes * 60 * 1000
    };
    this.storage.setItem(`${CACHE_KEY}:${key}`, JSON.stringify(entry));
  }
  
  // Cache parsed INP files
  cacheProject(fileName: string, project: SwmmProject): void {
    this.set(`project:${fileName}`, project, 120);
  }
  
  getCachedProject(fileName: string): SwmmProject | null {
    return this.get(`project:${fileName}`);
  }
  
  // Cache simulation results
  cacheResults(fileName: string, results: SimulationResults): void {
    // Don't cache large results in localStorage - use IndexedDB instead
    this.set(`results:${fileName}:meta`, {
      timeSteps: results.timeSteps.length,
      summary: results.summary
    }, 60);
  }
}
```

### IndexedDB for Large Result Storage

```typescript
// client/src/lib/cache/resultCache.ts
export class ResultCache {
  private db: IDBDatabase | null = null;
  
  async init(): Promise<void> {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open('SWMMResults', 1);
      
      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        this.db = request.result;
        resolve();
      };
      
      request.onupgradeneeded = (e) => {
        const db = (e.target as IDBOpenDBRequest).result;
        db.createObjectStore('results', { keyPath: 'id' });
        db.createObjectStore('timeSeries', { keyPath: 'id' });
      };
    });
  }
  
  async saveResults(id: string, results: SimulationResults): Promise<void> {
    if (!this.db) await this.init();
    
    // Store metadata separately from time series data
    const { timeSteps, ...metadata } = results;
    
    const tx = this.db!.transaction(['results', 'timeSeries'], 'readwrite');
    
    await Promise.all([
      this.putStore(tx.objectStore('results'), { id, ...metadata }),
      this.putStore(tx.objectStore('timeSeries'), { id, timeSteps })
    ]);
  }
  
  async getResults(id: string): Promise<SimulationResults | null> {
    if (!this.db) await this.init();
    
    const tx = this.db!.transaction(['results', 'timeSeries'], 'readonly');
    
    const [meta, ts] = await Promise.all([
      this.getStore(tx.objectStore('results'), id),
      this.getStore(tx.objectStore('timeSeries'), id)
    ]);
    
    if (!meta || !ts) return null;
    
    return {
      ...meta,
      timeSteps: ts.timeSteps
    };
  }
  
  private putStore(store: IDBObjectStore, data: any): Promise<void> {
    return new Promise((resolve, reject) => {
      const request = store.put(data);
      request.onsuccess = () => resolve();
      request.onerror = () => reject(request.error);
    });
  }
  
  private getStore(store: IDBObjectStore, id: string): Promise<any> {
    return new Promise((resolve, reject) => {
      const request = store.get(id);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }
}
```

### Add Keyboard Shortcuts System

```typescript
// client/src/hooks/useKeyboardShortcuts.ts
import { useEffect, useCallback } from 'react';

interface Shortcut {
  key: string;
  ctrl?: boolean;
  shift?: boolean;
  alt?: boolean;
  action: () => void;
  preventDefault?: boolean;
}

export function useKeyboardShortcuts(shortcuts: Shortcut[]) {
  const handler = useCallback((e: KeyboardEvent) => {
    shortcuts.forEach(s => {
      const keyMatch = e.key.toLowerCase() === s.key.toLowerCase();
      const ctrlMatch = !!s.ctrl === (e.ctrlKey || e.metaKey);
      const shiftMatch = !!s.shift === e.shiftKey;
      const altMatch = !!s.alt === e.altKey;
      
      if (keyMatch && ctrlMatch && shiftMatch && altMatch) {
        if (s.preventDefault !== false) e.preventDefault();
        s.action();
      }
    });
  }, [shortcuts]);
  
  useEffect(() => {
    window.addEventListener('keydown', handler);
    return () => window.removeEventListener('keydown', handler);
  }, [handler]);
}

// Usage in component
useKeyboardShortcuts([
  { key: 'n', ctrl: true, action: handleNew },
```
