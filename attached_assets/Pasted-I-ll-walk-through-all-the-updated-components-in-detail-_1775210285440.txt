```jsx
// src/components/PropertyGrid/PropertyGrid.jsx
import React, { useState, useCallback, useMemo } from 'react';
import './PropertyGrid.css';

// FieldEditor handles rendering and editing a single property value
const FieldEditor = ({ field, value, onChange, model }) => {
  const [editing, setEditing] = useState(false);
  const [tempValue, setTempValue] = useState(value);
  const [error, setError] = useState(null);

  // Validation runs before committing any edit
  const validate = (val) => {
    if (field.required && (val === '' || val === null || val === undefined)) {
      return 'Required field';
    }
    if (field.type === 'number') {
      const num = parseFloat(val);
      if (isNaN(num)) return 'Must be a number';
      if (field.min !== undefined && num < field.min) return `Min: ${field.min}`;
      if (field.max !== undefined && num > field.max) return `Max: ${field.max}`;
    }
    return null;
  };

  // Commit the edit — validates, then calls parent onChange
  const handleCommit = () => {
    const err = validate(tempValue);
    if (err) {
      setError(err);
      return;  // Don't commit invalid values
    }
    setError(null);
    setEditing(false);
    const finalValue = field.type === 'number' ? parseFloat(tempValue) : tempValue;
    onChange(field.key, finalValue);
    // This triggers: model.setProperty() → model._emit('change') → map redraw
  };

  const handleKeyDown = (e) => {
    if (e.key === 'Enter') handleCommit();
    if (e.key === 'Escape') {
      setTempValue(value);  // Revert
      setError(null);
      setEditing(false);
    }
  };

  // READONLY — just display, no editing (e.g., object Name after creation)
  if (field.type === 'readonly') {
    return <span className="pg-readonly">{value}</span>;
  }

  // ENUM — dropdown select (e.g., outfall Type: FREE/NORMAL/FIXED/TIDAL/TIMESERIES)
  if (field.type === 'enum') {
    return (
      <select
        className="pg-select"
        value={value || ''}
        onChange={(e) => onChange(field.key, e.target.value)}
      >
        {(field.enumValues || []).map(opt => (
          <option key={opt} value={opt}>{opt}</option>
        ))}
      </select>
    );
  }

  // BOOLEAN — checkbox toggle (e.g., Flap Gate: YES/NO)
  if (field.type === 'boolean') {
    return (
      <label className="pg-checkbox">
        <input
          type="checkbox"
          checked={value === true || value === 'YES'}
          onChange={(e) => onChange(field.key, e.target.checked ? 'YES' : 'NO')}
        />
        <span>{value === true || value === 'YES' ? 'YES' : 'NO'}</span>
      </label>
    );
  }

  // SUBDIALOG — shows summary + [...] button for complex editors
  // (e.g., Direct Inflows, LID Controls, Treatment)
  if (field.type === 'subdialog') {
    return (
      <div className="pg-subdialog">
        <span className="pg-subdialog-summary">{value || 'NONE'}</span>
        <button className="pg-subdialog-btn" title={`Edit ${field.name}`}>...</button>
      </div>
    );
  }

  // REF — dropdown populated from model data (e.g., Rain Gage selection)
  if (field.type === 'ref') {
    const options = field.getOptions ? field.getOptions(model) : [];
    return (
      <select className="pg-select" value={value || ''}
        onChange={(e) => onChange(field.key, e.target.value)}>
        <option value="">-- Select --</option>
        {options.map(opt => <option key={opt} value={opt}>{opt}</option>)}
      </select>
    );
  }

  // NUMBER or STRING — click-to-edit inline field
  if (editing) {
    return (
      <div className="pg-edit-wrapper">
        <input
          className={`pg-input ${error ? 'pg-input-error' : ''}`}
          type={field.type === 'number' ? 'number' : 'text'}
          value={tempValue ?? ''}
          onChange={(e) => { setTempValue(e.target.value); setError(null); }}
          onBlur={handleCommit}
          onKeyDown={handleKeyDown}
          autoFocus
          step={field.step || 'any'}
        />
        {field.units && <span className="pg-units">{field.units}</span>}
        {error && <div className="pg-error-tooltip">{error}</div>}
      </div>
    );
  }

  // Display mode — click to start editing
  return (
    <div className="pg-value-display"
      onClick={() => { setTempValue(value); setEditing(true); }}
      tabIndex={0}
      onFocus={() => { setTempValue(value); setEditing(true); }}>
      <span>{value ?? ''}</span>
      {field.units && <span className="pg-units">{field.units}</span>}
    </div>
  );
};

// Main PropertyGrid component
const PropertyGrid = ({ objectType, objectId, data, schema, onChange, 
                        onOpenSubEditor, model, onClose }) => {
  const [collapsed, setCollapsed] = useState({});

  const toggleCategory = (cat) => {
    setCollapsed(prev => ({ ...prev, [cat]: !prev[cat] }));
  };

  const handleFieldChange = useCallback((key, value) => {
    if (onChange) onChange(key, value);
  }, [onChange]);

  // Group fields by category
  const categories = useMemo(() => {
    const cats = [];
    const catMap = {};
    (schema || []).forEach(field => {
      const cat = field.category || 'General';
      if (!catMap[cat]) { catMap[cat] = []; cats.push(cat); }
      catMap[cat].push(field);
    });
    return cats.map(c => ({ name: c, fields: catMap[c] }));
  }, [schema]);

  // Empty state when nothing selected
  if (!objectId || !data) {
    return (
      <div className="property-grid">
        <div className="pg-header"><span>Property Editor</span></div>
        <div className="pg-empty">
          Click an object on the map or project tree to edit its properties.
        </div>
      </div>
    );
  }

  const typeLabel = objectType ? 
    objectType.charAt(0).toUpperCase() + objectType.slice(1) : '';

  return (
    <div className="property-grid">
      <div className="pg-header">
        <span className="pg-title">Property Editor</span>
        <div className="pg-header-actions">
          <button className="pg-pin" title="Pin">📌</button>
          <button className="pg-close" onClick={onClose} title="Close">✕</button>
        </div>
      </div>

      {/* Object type badge + ID */}
      <div className="pg-object-banner">
        <span className="pg-object-type">{typeLabel}</span>
        <span className="pg-object-id">{objectId}</span>
      </div>

      {/* Scrollable field list grouped by category */}
      <div className="pg-body">
        {categories.map(cat => (
          <div key={cat.name} className="pg-category">
            <div className="pg-category-header" 
                 onClick={() => toggleCategory(cat.name)}>
              <span className="pg-collapse-icon">
                {collapsed[cat.name] ? '▶' : '▼'}
              </span>
              <span className="pg-category-name">{cat.name}</span>
            </div>
            {!collapsed[cat.name] && (
              <table className="pg-fields-table">
                <tbody>
                  {cat.fields.map(field => {
                    // CONDITIONAL VISIBILITY — key feature!
                    // e.g., "Fixed Stage" only shows when outfallType === "FIXED"
                    if (field.visibleWhen) {
                      const condField = field.visibleWhen.field;
                      const condValues = field.visibleWhen.values;
                      if (!condValues.includes(data[condField])) {
                        return null;  // Hide this field
                      }
                    }
                    return (
                      <tr key={field.key} className="pg-field-row">
                        <td className="pg-field-name" title={field.helpText || ''}>
                          {field.name}
                          {field.required && <span className="pg-required">*</span>}
                        </td>
                        <td className="pg-field-value">
                          <FieldEditor
                            field={field}
                            value={data[field.key]}
                            onChange={handleFieldChange}
                            model={model}
                          />
                        </td>
                      </tr>
                    );
                  })}
                </tbody>
              </table>
            )}
          </div>
        ))}
      </div>
    </div>
  );
};

export default PropertyGrid;
```

## 2. SWMM MODEL — Parser & Serializer

```javascript
// src/model/SWMMModel.js — Core data model
// Handles: .inp parsing, property-level CRUD, .inp serialization, event system

class SWMMModel {
  constructor() {
    // Data stores for every SWMM object type
    this.junctions = new Map();    // 149 in Greenville model
    this.outfalls = new Map();     // 5
    this.storage = new Map();      // 17
    this.dividers = new Map();     // 1
    this.conduits = new Map();     // 223
    this.pumps = new Map();        // 5
    this.orifices = new Map();     // 2
    this.weirs = new Map();        // 4
    this.outlets = new Map();      // 1
    this.subcatchments = new Map(); // 30
    this.raingages = new Map();
    this.timeseries = new Map();
    this.curves = new Map();
    this.patterns = new Map();
    this.xsections = new Map();    // 229
    this.losses = new Map();       // 223
    this.coordinates = new Map();
    this.vertices = new Map();
    this.polygons = new Map();
    this.symbols = new Map();
    this.tags = new Map();
    this.inflows = new Map();
    this.dwf = new Map();
    this.controls = '';
    this.options = {};
    
    // Round-trip fidelity
    this._rawSections = {};    // Preserves original text for unknown sections
    this._sectionOrder = [];   // Preserves original section ordering
    this._listeners = {};      // Event system
    this._dirty = false;       // Tracks unsaved changes
  }

  // ─── EVENT SYSTEM ──────────────────────────────────
  // Used to trigger map redraws and auto-save when properties change
  
  on(event, callback) {
    if (!this._listeners[event]) this._listeners[event] = [];
    this._listeners[event].push(callback);
  }
  
  _emit(event, data) {
    (this._listeners[event] || []).forEach(cb => cb(data));
  }

  // ─── CRUD OPERATIONS ──────────────────────────────
  
  getObject(type, id) {
    const store = this._getStore(type);
    return store ? store.get(id) : null;
  }

  setProperty(type, id, key, value) {
    const store = this._getStore(type);
    if (!store) return;
    const obj = store.get(id);
    if (!obj) return;
    const oldValue = obj[key];
    obj[key] = value;
    this._dirty = true;
    // This event triggers map redraw + auto-save
    this._emit('change', { type, id, key, value, oldValue });
  }

  addObject(type, data) {
    const store = this._getStore(type);
    if (!store || !data.name) return null;
    store.set(data.name, { ...data });
    this._dirty = true;
    this._emit('add', { type, id: data.name, data });
    return data.name;
  }

  deleteObject(type, id) {
    const store = this._getStore(type);
    if (!store) return false;
    const existed = store.delete(id);
    if (existed) {
      this._dirty = true;
      this._emit('delete', { type, id });
    }
    return existed;
  }

  // ─── .inp PARSER ───────────────────────────────────
  // Reads the standard EPA SWMM5 input file format
  
  fromINP(text) {
    const lines = text.split(/\r?\n/);
    let currentSection = null;
    this._sectionOrder = [];
    this._rawSections = {};

    // First pass: split into sections
    for (const line of lines) {
      const trimmed = line.trim();
      const sectionMatch = trimmed.match(/^\[(\w+)\]$/);
      if (sectionMatch) {
        currentSection = sectionMatch[1].toUpperCase();
        if (!this._sectionOrder.includes(currentSection)) {
          this._sectionOrder.push(currentSection);
        }
        if (!this._rawSections[currentSection]) {
          this._rawSections[currentSection] = [];
        }
        continue;
      }
      if (currentSection) {
        this._rawSections[currentSection].push(line);
      }
    }

    // Second pass: parse each known section
    this._parseJunctions();   // [JUNCTIONS]
    this._parseOutfalls();    // [OUTFALLS]
    this._parseStorage();     // [STORAGE]
    this._parseConduits();    // [CONDUITS]
    this._parseXsections();   // [XSECTIONS] — attaches to conduits
    this._parseLosses();      // [LOSSES] — attaches to conduits
    this._parseSubcatchments(); // [SUBCATCHMENTS]
    this._parseSubareas();    // [SUBAREAS] — attaches to subcatchments
    this._parseInfiltration(); // [INFILTRATION] — attaches to subcatchments
    this._parseCoordinates(); // [COORDINATES] — attaches to nodes
    this._parsePolygons();    // [POLYGONS] — subcatchment shapes
    // ... plus all other sections
  }

  // Example section parser:
  _parseJunctions() {
    for (const line of this._getDataLines('JUNCTIONS')) {
      const p = this._splitFields(line);
      if (p.length >= 2) {
        this.junctions.set(p[0], {
          name: p[0],
          invertElev: parseFloat(p[1]) || 0,
          maxDepth: parseFloat(p[2]) || 0,
          initDepth: parseFloat(p[3]) || 0,
          surchargeDepth: parseFloat(p[4]) || 0,
          pondedArea: parseFloat(p[5]) || 0,
        });
      }
    }
  }

  // Helper: skip comment lines (starting with ;;)
  _getDataLines(section) {
    return (this._rawSections[section] || [])
      .filter(l => {
        const t = l.trim();
        return t && !t.startsWith(';;') && !t.startsWith(';');
      });
  }

  // ─── .inp SERIALIZER ──────────────────────────────
  // Writes back to .inp format preserving original structure
  
  toINP() {
    const lines = [];
    
    // Process sections in ORIGINAL order (round-trip fidelity)
    for (const section of this._sectionOrder) {
      if (knownSections.has(section)) {
        this._writeKnownSection(section, lines);
      } else {
        // Unknown sections pass through VERBATIM
        lines.push(`[${section}]`);
        (this._rawSections[section] || []).forEach(l => lines.push(l));
        lines.push('');
      }
    }
    return lines.join('\n');
  }
}
```

```javascript
// src/components/ResultsAnimation/SWMMOutputReader.js
// Reads SWMM5's native binary output format directly in JavaScript

class SWMMOutputReader {
  constructor(arrayBuffer) {
    this.buffer = arrayBuffer;
    this.view = new DataView(arrayBuffer);
    this.offset = 0;
    
    // These get populated during parsing
    this.subcatchNames = [];
    this.nodeNames = [];
    this.linkNames = [];
    this.nPeriods = 0;
    this.reportStep = 300; // seconds
    this.bytesPerPeriod = 0;
    this.resultOffset = 0;
    
    this._parse();
  }

  _readInt() {
    const val = this.view.getInt32(this.offset, true); // little-endian
    this.offset += 4;
    return val;
  }

  _readFloat() {
    const val = this.view.getFloat32(this.offset, true);
    this.offset += 4;
    return val;
  }

  _parse() {
    // HEADER: magic(4), version(4), flowUnits(4), counts(4×4)
    const magic1 = this._readInt();    // Should be 516114522
    const version = this._readInt();
    const flowUnits = this._readInt();
    this.nSubcatch = this._readInt();
    this.nNodes = this._readInt();
    this.nLinks = this._readInt();
    this.nPolluts = this._readInt();

    // ID STRINGS: variable-length strings for each object
    for (let i = 0; i < this.nSubcatch; i++) 
      this.subcatchNames.push(this._readString());
    for (let i = 0; i < this.nNodes; i++) 
      this.nodeNames.push(this._readString());
    for (let i = 0; i < this.nLinks; i++) 
      this.linkNames.push(this._readString());

    // Skip object properties and reporting variable definitions
    // ... (see full implementation)

    // Record where the actual results data starts
    this.resultOffset = this.offset;

    // Calculate bytes per timestep
    this.bytesPerPeriod = 8 +  // date double
      (this.nSubcatch * this.nSubcatchVars * 4) +
      (this.nNodes * this.nNodeVars * 4) +
      (this.nLinks * this.nLinkVars * 4) +
      (this.nSystemVars * 4);

    // Read footer for period count
    const footerOffset = this.buffer.byteLength - 12;
    const footerView = new DataView(this.buffer, footerOffset);
    this.nPeriods = footerView.getInt32(0, true);
  }

  // ─── DATA ACCESS ──────────────────────────────────
  
  // Get a single value: e.g., Node J-42 depth at timestep 15
  getNodeResult(nodeId, variable, timestep) {
    const idx = this.nodeNames.indexOf(nodeId);
    const varIdx = NODE_RESULT_VARS.indexOf(variable);
    if (idx < 0 || varIdx < 0) return 0;
    return this._getResultValue(timestep, null, idx, null, varIdx);
  }

  // Get full time series: e.g., all depths at Node J-42
  getNodeSeries(nodeId, variable) {
    const series = [];
    for (let t = 0; t < this.nPeriods; t++) {
      series.push({
        time: t * this.reportStep,
        value: this.getNodeResult(nodeId, variable, t),
      });
    }
    return series;
  }

  // Get value range for legend coloring at a timestep
  getLegendRange(objectType, variable, timestep) {
    let min = Infinity, max = -Infinity;
    const names = objectType === 'node' ? this.nodeNames : 
                  objectType === 'link' ? this.linkNames : 
                  this.subcatchNames;
    const getter = objectType === 'node' ? 'getNodeResult' : 
                   objectType === 'link' ? 'getLinkResult' : 
                   'getSubcatchResult';
    
    for (const name of names) {
      const val = this[getter](name, variable, timestep);
      if (val < min) min = val;
      if (val > max) max = val;
    }
    return { min: min === Infinity ? 0 : min, 
             max: max === -Infinity ? 1 : max };
  }
}
```

```jsx
function detectErrors(model) {
  const issues = [];
  
  // Junction checks
  model.junctions.forEach((j, id) => {
    if (j.invertElev === undefined)
      issues.push({ severity: 'error', objectId: id, 
        message: 'Missing invert elevation' });
    if (j.maxDepth > 50)
      issues.push({ severity: 'warning', objectId: id, 
        message: `Unusually large max depth: ${j.maxDepth} ft` });
  });

  // Conduit checks
  model.conduits.forEach((c, id) => {
    if (c.fromNode === c.toNode)
      issues.push({ severity: 'error', objectId: id, 
        message: 'Self-loop conduit' });
    if (c.roughness < 0.005 || c.roughness > 0.5)
      issues.push({ severity: 'warning', objectId: id, 
        message: `Unusual Manning's n: ${c.roughness}`,
        fix: 'Use 0.013 for concrete, 0.024 for corrugated metal' });
    
    // Slope check using connected node elevations
    const slope = (fromInvert - toInvert) / c.length;
    if (slope < 0)
      issues.push({ severity: 'warning', objectId: id, 
        message: `Adverse slope: ${(slope*100).toFixed(3)}%` });
  });

  // Connectivity: find orphan nodes
  model.junctions.forEach((_, id) => {
    if (!connectedNodes.has(id))
      issues.push({ severity: 'warning', objectId: id, 
        message: 'Orphan node — not connected to any link' });
  });
  
  return issues;
}

function estimateConduitLength(conduit, model) {
  const from = model.coordinates.get(conduit.fromNode);
  const to = model.coordinates.get(conduit.toNode);
```
